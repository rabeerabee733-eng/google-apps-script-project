/*********************** * ضبط ملف Google Sheets ***********************/ var SHEET_ID = '1He3MxPqQGcoOgUGEPoI1iVX6L5-U7j2TjQll2dWhr_w'; /*********************** * doGet (للاختبار فقط) ***********************/ function doGet(e) { return ContentService.createTextOutput('OK'); } /**************************************** * حفظ الطلب في Google Sheets ****************************************/ function saveOrder(order) { order = order || {}; try { var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheets()[0]; // إنشاء عناوين الأعمدة إذا كان الشيت فارغاً if (sheet.getLastRow() === 0) { sheet.appendRow([ 'الرقم', 'التاريخ والوقت', 'الاسم', 'الهاتف الأساسي', 'هاتف بديل', 'المحافظة', 'اللواء', 'العنوان التفصيلي' ]); } var orderNumber = sheet.getLastRow(); // رقم الطلب sheet.appendRow([ orderNumber, new Date(), order.name || '', order.phone || '', order.altPhone || '', order.governorate || '', order.district || '', order.address || '' ]); return orderNumber; } catch (err) { Logger.log(err); throw err; } } /**************************************** * doPost – تستقبل الطلب من GitHub ****************************************/ function doPost(e) { var result; try { var dataString = ''; if (e && e.parameter && e.parameter.data) { dataString = e.parameter.data; } else if (e && e.postData && e.postData.contents) { dataString = e.postData.contents; } else { throw new Error('No data found in request'); } var order = JSON.parse(dataString); var orderNumber = saveOrder(order); result = { success: true, orderNumber: orderNumber }; } catch (err) { Logger.log(err); result = { success: false, error: err.toString() }; } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
